42// 默认的问题实现，使用配置文件中的约束来定义各种算子
43impl 默认操作 {
44    pub fn 新建(上下文: &默认上下文) -> Result<Self, 错误> {
45        Ok(Self {
46            决策空间: 上下文.决策空间.clone(),
47            元素图: 上下文.元素图.clone(),
48        })
49    }
50
51    fn 传播(&self, 变化: &mut <默认决策 as 决策>::变化, 决策: &mut 默认决策) {
52        // 初始化队列
53        let mut 队列 = VecDeque::new();
54        for 元素 in 变化
55            .增加元素
56            .iter()
57            .chain(变化.减少元素.iter())
58            .chain(变化.移动元素.iter())
59        {
60            for 下游元素 in self.元素图.get(元素).unwrap_or(&vec![]) {
61                if !队列.contains(下游元素) {
62                    队列.push_back(下游元素.clone());
63                }
64            }
65        }
66        // 传播直到队列为空
67        let mut iters = 0;
68        while !队列.is_empty() {
69            iters += 1;
70            if iters > 100 {
71                panic!("传播超过 100 次仍未结束，可能出现死循环");
72            }
73            let 元素 = 队列.pop_front().unwrap();
74            let 当前安排 = 决策.元素[元素];
75            let mut 合法 = false;
76            let mut 新安排列表 = vec![];
77            for 条件安排 in &self.决策空间.元素[元素] {
78                if 决策.允许(条件安排) {
79                    if 条件安排.安排 == 当前安排 {
80                        合法 = true;
81                        break;
82                    }
83                    新安排列表.push(条件安排.安排.clone());
84                }
85            }
86            if !合法 {
87                if 新安排列表.is_empty() {
88                    panic!("没有合法的安排，传播失败");
89                } else {
90                    let 新安排 = *新安排列表.choose(&mut rng()).unwrap();
91                    if let 默认安排::未选取 = 当前安排 {
92                        变化.增加元素.push(元素);
93                    } else if let 默认安排::未选取 = 新安排 {
94                        变化.减少元素.push(元素);
95                    } else {
96                        变化.移动元素.push(元素);
97                    }
98                    决策.元素[元素] = 新安排;
99                }
100            }
101            for 下游元素 in self.元素图.get(&元素).unwrap_or(&vec![]) {
102                if !队列.contains(下游元素) {
103                    队列.push_back(*下游元素);
104                }
105            }
106        }
107    }
108
109    pub fn 随机移动元素(&self, 决策: &mut 默认决策) -> 默认决策变化 {
110        let mut rng = rng();
111        const MAX_TRIES: usize = 100;
112        for _ in 0..MAX_TRIES {
113            let 元素 = (0..决策.元素.len()).choose(&mut rng).unwrap();
114            let 当前安排 = 决策.元素[元素];
115            // 蓄水池抽样
116            let mut 下一个安排 = None;
117            let mut count = 0;
118            for 条件安排 in &self.决策空间.元素[元素] {
119                if 条件安排.安排 != 当前安排 && 决策.允许(条件安排) {
120                    count += 1;
121                    if random_range(0..count) == 0 {
122                        下一个安排 = Some(&条件安排.安排);
123                    }
124                }
125            }
126            if let Some(下一个安排) = 下一个安排 {
127                决策.元素[元素] = *下一个安排;
128                let mut 增加元素 = vec![];
129                let mut 减少元素 = vec![];
130                let mut 移动元素 = vec![];
131                if let 默认安排::未选取 = 当前安排 {
132                    增加元素.push(元素);
133                } else if let 默认安排::未选取 = 下一个安排 {
134                    减少元素.push(元素);
135                } else {
136                    移动元素.push(元素);
137                }
138                return 默认决策变化::新建(增加元素, 减少元素, 移动元素);
139            }
140        }
141        默认决策变化::不变()
142    }
