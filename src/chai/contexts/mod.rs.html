31pub fn 拓扑排序(
32    原始决策空间: &IndexMap<String, Vec<安排描述>>,
33) -> Result<(Vec<String>, FxHashMap<String, Vec<String>>), 错误> {
34    // 构造入度表
35    let mut 入度 = FxHashMap::default();
36    let mut 元素图 = FxHashMap::default();
37    for 元素名称 in 原始决策空间.keys() {
38        入度.insert(元素名称.clone(), 0);
39        元素图.insert(元素名称.clone(), vec![]);
40    }
41    for (元素名称, 原始安排列表) in 原始决策空间 {
42        let mut 依赖 = FxHashSet::default();
43        for 原始安排 in 原始安排列表 {
44            if let 安排::Advanced(keys) = &原始安排.value {
45                for k in keys {
46                    if let 广义码位::Reference { element, .. } = k {
47                        依赖.insert(element.clone());
48                    }
49                }
50            } else if let 安排::Grouped { element } = &原始安排.value {
51                依赖.insert(element.clone());
52            }
53            if let Some(条件列表) = &原始安排.condition {
54                for 条件 in 条件列表 {
55                    依赖.insert(条件.element.clone());
56                }
57            }
58        }
59        for 依赖元素 in &依赖 {
60            元素图.get_mut(依赖元素).map(|v| {
61                v.push(元素名称.clone());
62                *入度.get_mut(元素名称).unwrap() += 1;
63            });
64        }
65    }
66
67    // 拓扑排序
68    let mut 队列 = VecDeque::new();
69    for (元素名称, d) in &入度 {
70        if *d == 0 {
71            队列.push_back(元素名称.clone());
72        }
73    }
74
75    let mut 排序后元素名称 = Vec::new();
76    while let Some(u) = 队列.pop_front() {
77        排序后元素名称.push(u.clone());
78        for v in &元素图[&u] {
79            let deg = 入度.get_mut(v).unwrap();
80            *deg -= 1;
81            if *deg == 0 {
82                队列.push_back(v.clone());
83            }
84        }
85    }
86
87    // 检测环
88    if 排序后元素名称.len() != 原始决策空间.len() {
89        let remaining: Vec<_> = 入度
90            .into_iter()
91            .filter(|(_, deg)| *deg > 0)
92            .map(|(k, _)| k)
93            .collect();
94        return Err(format!("检测到依赖环，无法进行拓扑排序，剩余节点：{:?}", remaining).into());
95    }
96
97    Ok((排序后元素名称, 元素图))
98}
